const VSCODE_SHIM_FOR_CALLER = '_VSCODE_IMPORT_VSCODE_API_FOR_CALLER';

/**
 * vshell/codebun: installs a real `vscode` package for runtimes that cannot intercept
 * module resolution (Bun has no `module.registerHooks` and ignores `module._load`).
 *
 * Node resolution is honoured by every runtime, so a `node_modules/vscode` next to the
 * extensions is reachable from all of them. The package hands the requiring file name
 * back to us, which is the one thing the hooks provided and a plain module cannot know:
 * the API instance is per extension.
 *
 * The generated module is CommonJS (`module.exports = instance`), which covers
 * `require('vscode')` and `import vscode from 'vscode'`. A namespace import
 * (`import * as vscode`) would rely on statically detectable named exports and so needs
 * the hook path; no extension shipped with this build uses one.
 */
function installVscodeShimPackage(initData: IExtHostInitDataService, apiInstanceFor: (url: string) => unknown): IDisposable {
	const fs = require('fs') as typeof import('fs');
	const path = require('path') as typeof import('path');
	const allExtensions = initData.extensions?.allExtensions ?? [];
	if (allExtensions.length === 0) {
		return toDisposable(() => { /* nothing to intercept */ });
	}

	// Built-in and user extensions live in different trees, and module resolution only walks
	// up from the extension's own folder, so every root needs its own copy of the package.
	const roots = new Set<string>();
	for (const extension of allExtensions) {
		roots.add(path.dirname(extension.extensionLocation.fsPath));
	}

	const packageJson = '{\n\t"name": "vscode",\n\t"version": "0.0.0",\n\t"main": "index.js"\n}\n';
	const indexJs = [
		'"use strict";',
		'// Generated by vshell/codebun at startup. Do not edit: it is rewritten on every launch.',
		'// Bun cannot intercept module resolution, so `vscode` is a real package that asks the',
		'// extension host for the API instance belonging to whoever required it.',
		`const apiForCaller = globalThis['${VSCODE_SHIM_FOR_CALLER}'];`,
		'if (typeof apiForCaller !== \'function\') {',
		'\tthrow new Error(\'vscode API shim was loaded without the extension host\');',
		'}',
		'module.exports = apiForCaller(module.parent && module.parent.filename);',
		'',
	].join('\n');
	const files: [string, string][] = [['package.json', packageJson], ['index.js', indexJs]];

	for (const root of roots) {
		const shimDir = path.join(root, 'node_modules', 'vscode');
		try {
			fs.mkdirSync(shimDir, { recursive: true });
			for (const [name, content] of files) {
				const file = path.join(shimDir, name);
				if (!fs.existsSync(file) || fs.readFileSync(file, 'utf8') !== content) {
					fs.writeFileSync(file, content);
				}
			}
		} catch (err) {
			// Activation is worth more than the shim: report loudly and keep going, so one
			// read-only extension root cannot take the whole extension host down.
			console.error(`[vshell] could not write the vscode module shim in ${shimDir}: ${err}`);
		}
	}

	Object.defineProperty(globalThis, VSCODE_SHIM_FOR_CALLER, {
		enumerable: false,
		configurable: true,
		writable: false,
		value: (callerFilename: string | undefined) => {
			if (!callerFilename) {
				throw new Error('vscode API shim: the requiring module has no file name');
			}
			let caller = callerFilename;
			try {
				caller = realpathSync(callerFilename);
			} catch {
				// keep the unresolved path: the API factory does its own containment lookup
			}
			return apiInstanceFor(URI.file(caller).toString());
		}
	});

	return toDisposable(() => {
		delete (globalThis as { [key: string]: unknown })[VSCODE_SHIM_FOR_CALLER];
	});
}
